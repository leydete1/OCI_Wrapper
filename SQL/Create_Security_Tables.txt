CREATE TABLE AUTH_SOURCE (
    AUTH_SOURCE_ID     NUMBER          GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    NAME               VARCHAR2(100)   NOT NULL,
    SOURCE_TYPE        VARCHAR2(20)    NOT NULL
                          CHECK (SOURCE_TYPE IN ('LOCAL', 'LDAP', 'AD')),
    CONFIGURATION      CLOB,            -- JSON: host, port, base_dn, bind pattern, etc.
                                        -- NULL / ignored when SOURCE_TYPE = 'LOCAL'
    ENABLED            VARCHAR2(1)     DEFAULT 'Y' NOT NULL
                          CHECK (ENABLED IN ('Y', 'N')),
    CREATED_TS         TIMESTAMP       DEFAULT SYSTIMESTAMP NOT NULL,
    MODIFIED_TS        TIMESTAMP       DEFAULT SYSTIMESTAMP NOT NULL
);
 
CREATE UNIQUE INDEX IDX_AUTH_SOURCE_NAME_UQ ON AUTH_SOURCE (NAME);
 
-- Seed row required by every deployment – local auth must always be
-- available even before any LDAP/AD source is configured.
INSERT INTO AUTH_SOURCE (NAME, SOURCE_TYPE, ENABLED)
    VALUES ('Local Database', 'LOCAL', 'Y');
    
    
    CREATE TABLE APP_USER (
    USER_ID            NUMBER          GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    USERNAME           VARCHAR2(128)   NOT NULL,
    DISPLAY_NAME       VARCHAR2(255),
    AUTH_SOURCE_ID     NUMBER          NOT NULL
                          REFERENCES AUTH_SOURCE (AUTH_SOURCE_ID),
 
    -- Only meaningful when AUTH_SOURCE_ID points at a LOCAL row.
    -- Never populated for LDAP/AD users.
    PASSWORD_HASH      VARCHAR2(255),
    PASSWORD_PARAMS    VARCHAR2(255),   -- Argon2id params string (see 6.5)
 
    ENABLED            VARCHAR2(1)     DEFAULT 'Y' NOT NULL
                          CHECK (ENABLED IN ('Y', 'N')),
    LOCKED             VARCHAR2(1)     DEFAULT 'N' NOT NULL
                          CHECK (LOCKED IN ('Y', 'N')),
    FAILED_ATTEMPTS    NUMBER          DEFAULT 0 NOT NULL,
    LOCKED_TS          TIMESTAMP,
 
    LAST_LOGIN_TS      TIMESTAMP,
    CREATED_TS         TIMESTAMP       DEFAULT SYSTIMESTAMP NOT NULL,
    MODIFIED_TS        TIMESTAMP       DEFAULT SYSTIMESTAMP NOT NULL
);
 
CREATE UNIQUE INDEX IDX_APP_USER_USERNAME_UQ ON APP_USER (UPPER(USERNAME));
CREATE INDEX IDX_APP_USER_AUTH_SOURCE ON APP_USER (AUTH_SOURCE_ID);


CREATE TABLE ROLE (
    ROLE_ID            NUMBER          GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    ROLE_NAME          VARCHAR2(100)   NOT NULL,
    DESCRIPTION        VARCHAR2(500)
);
CREATE UNIQUE INDEX IDX_ROLE_NAME_UQ ON ROLE (ROLE_NAME);
 
CREATE TABLE PERMISSION (
    PERMISSION_ID      NUMBER          GENERATED ALWAYS AS IDENTITY PRIMARY KEY,
    PERMISSION_CODE    VARCHAR2(100)   NOT NULL,   -- e.g. 'CUSTOMER.READ'
    DESCRIPTION        VARCHAR2(500)
);
CREATE UNIQUE INDEX IDX_PERMISSION_CODE_UQ ON PERMISSION (PERMISSION_CODE);
 
CREATE TABLE USER_ROLE (
    USER_ID            NUMBER  NOT NULL REFERENCES APP_USER (USER_ID),
    ROLE_ID             NUMBER  NOT NULL REFERENCES ROLE (ROLE_ID),
    PRIMARY KEY (USER_ID, ROLE_ID)
);
 
CREATE TABLE ROLE_PERMISSION (
    ROLE_ID             NUMBER  NOT NULL REFERENCES ROLE (ROLE_ID),
    PERMISSION_ID       NUMBER  NOT NULL REFERENCES PERMISSION (PERMISSION_ID),
    PRIMARY KEY (ROLE_ID, PERMISSION_ID)
);
