CREATE TABLE DATA_MANAGER.UNIT_TEST_FIELD_TEST (
    NUMBER_COL      NUMBER          NOT NULL,
    VARCHAR2_COL    VARCHAR2(100),
    DATE_COL        DATE,
    CLOB_COL        CLOB,
    CONSTRAINT UNIT_TEST_FIELD_TEST_PK PRIMARY KEY (NUMBER_COL)
);


CREATE OR REPLACE PROCEDURE DATA_MANAGER.UNIT_TEST_SAMPLE_PROC (
    P_IN_NUM    IN  NUMBER,
    P_OUT_NUM   OUT NUMBER,
    P_RESULTS   OUT SYS_REFCURSOR
) AS
BEGIN
    P_OUT_NUM := P_IN_NUM * 2;

    OPEN P_RESULTS FOR
        SELECT NUMBER_COL, VARCHAR2_COL, DATE_COL
        FROM   DATA_MANAGER.UNIT_TEST_FIELD_TEST
        WHERE  NUMBER_COL = P_IN_NUM;
END;
/


CREATE OR REPLACE TRIGGER DATA_MANAGER.UNIT_TEST_FIELD_TEST_DEL_BLOCK
BEFORE DELETE ON DATA_MANAGER.UNIT_TEST_FIELD_TEST
FOR EACH ROW
WHEN (OLD.NUMBER_COL = 999099)
BEGIN
    RAISE_APPLICATION_ERROR(-20001, 'UT-DEL-003: DELETE deliberately blocked for testing');
END;
/


CREATE OR REPLACE PROCEDURE DATA_MANAGER.UNIT_TEST_NOOP_PROC AS
BEGIN
    NULL;
END;
/

CREATE OR REPLACE PROCEDURE DATA_MANAGER.UNIT_TEST_CURSOR_PROC (
    P_OPEN_CURSOR IN  NUMBER,
    P_RESULTS     OUT SYS_REFCURSOR
) AS
BEGIN
    IF P_OPEN_CURSOR = 1 THEN
        OPEN P_RESULTS FOR
            SELECT NUMBER_COL FROM DATA_MANAGER.UNIT_TEST_FIELD_TEST
            WHERE NUMBER_COL = 1;
    END IF;
    -- P_RESULTS stays genuinely unopened (NULL) when P_OPEN_CURSOR != 1
END;
/
